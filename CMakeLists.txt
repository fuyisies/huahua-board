cmake_minimum_required(VERSION 3.21)
project(HuahuaBoard LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
add_executable(HuahuaBoard WIN32 src/main.cpp assets/app.rc)
target_include_directories(HuahuaBoard PRIVATE assets)
target_compile_definitions(HuahuaBoard PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(HuahuaBoard PRIVATE d3d11 dxgi d2d1 dwrite windowscodecs ole32 uuid shell32 comdlg32 dwmapi gdi32)
set_target_properties(HuahuaBoard PROPERTIES OUTPUT_NAME "画画图板")
if(MSVC)
 set_property(TARGET HuahuaBoard PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 target_compile_options(HuahuaBoard PRIVATE /utf-8 /EHsc)
else()
 target_link_options(HuahuaBoard PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
